    user='root', password='',
    charset='utf8mb4', autocommit=True,
)
cur = conn.cursor()

cur.execute("SELECT * FROM users")
print(cur.fetchall())


SQL 支持
-- DDL
CREATE TABLE users (id INT PRIMARY KEY, name VARCHAR(50), age INT)
DROP TABLE users
CREATE INDEX idx_name ON users (name)

-- DML
INSERT INTO users VALUES (1, 'Alice', 30)
UPDATE users SET age = 31 WHERE id = 1
DELETE FROM users WHERE id = 1

-- 查询
SELECT * FROM users
SELECT * FROM users WHERE age > 20 ORDER BY age DESC LIMIT 10
SELECT city, COUNT(*) FROM users GROUP BY city HAVING COUNT(*) > 5
SELECT DISTINCT city FROM users

-- JOIN
SELECT u.name, o.amount FROM users u
  JOIN orders o ON u.id = o.user_id

-- 子查询
SELECT * FROM users WHERE id IN (SELECT user_id FROM orders)
SELECT * FROM users WHERE EXISTS (SELECT 1 FROM orders)

-- UNION
SELECT name FROM users UNION SELECT name FROM admins

-- 事务
BEGIN;
INSERT INTO users VALUES (2, 'Bob', 25);
COMMIT;
ROLLBACK;

-- 权限
CREATE USER 'alice' IDENTIFIED BY 'pass123'
GRANT SELECT, INSERT ON *.* TO 'alice'
REVOKE INSERT ON *.* FROM 'alice'
DROP USER 'alice'

-- 优化
ANALYZE TABLE users
EXPLAIN SELECT * FROM users WHERE id = 100
SHOW STATS
